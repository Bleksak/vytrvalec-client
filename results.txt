F(n) =
1 -> f1: (10,1), f2: (20,2), f3: (30,3)
2 -> f1: (30,3), f2: (20,2), f3: (30,3)
3 -> f1: (10,2), f2: (20,3), f3: (20,3)


f(n) =
1 -> f1: (10,1), f2: (20,2), f3: (30,3)
2 -> f1: (40,4), f2: (40,4), f3: (60,3)
3 -> f1: (50,6), f2: (60,7), f3: (80,6)

Pro get:
f(0) = F(0)
F(n) = f(n) - f(n-1)

Pro generaci:
f(n) = f(n-1) + F(n)
